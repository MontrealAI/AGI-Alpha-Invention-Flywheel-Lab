"""Reassemble bounded transport parts into the exact raw evidence ZIP; never extract or execute it."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import stat

MAX_PART_BYTES = 6 * 1024**2
MAX_ARCHIVE_BYTES = 64 * 1024**2
HEX = re.compile(r'[0-9a-f]{64}')


def safe(path):
    path = Path(path).absolute()
    if path.is_symlink() or any(parent.is_symlink() for parent in path.parents):
        raise ValueError('TRANSPORT_SYMLINK_REFUSED')
    return path


def inspect(path, expected_bytes, expected_sha, destination=None):
    path = safe(path)
    descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    digest, size = hashlib.sha256(), 0
    with os.fdopen(descriptor, 'rb') as source:
        metadata = os.fstat(source.fileno())
        if not stat.S_ISREG(metadata.st_mode) or metadata.st_size != expected_bytes:
            raise ValueError('TRANSPORT_PART_SIZE_MISMATCH')
        while True:
            block = source.read(65536)
            if not block:
                break
            size += len(block)
            if size > expected_bytes:
                raise ValueError('TRANSPORT_PART_SIZE_MISMATCH')
            digest.update(block)
            if destination is not None:
                destination.write(block)
    if size != expected_bytes or digest.hexdigest() != expected_sha:
        raise ValueError('TRANSPORT_PART_IDENTITY_MISMATCH')


def reassemble(index_path, output, expected_sha256):
    index_path, output = safe(index_path), safe(output)
    if not HEX.fullmatch(expected_sha256):
        raise ValueError('TRUSTED_WHOLE_SHA256_REQUIRED')
    if output.exists() or output.is_symlink():
        raise FileExistsError(output)
    if not output.parent.is_dir() or index_path.stat().st_size > 65536:
        raise ValueError('TRANSPORT_PATH_OR_INDEX_LIMIT')
    index = json.loads(index_path.read_text())
    required = {'schema', 'archive_file', 'archive_sha256', 'archive_bytes', 'parts'}
    if type(index) is not dict or set(index) != required or index['schema'] != 'agi-alpha/raw-evidence-multipart006/v1':
        raise ValueError('TRANSPORT_INDEX_SCHEMA')
    if index['archive_sha256'] != expected_sha256 or type(index['archive_bytes']) is not int or not 0 < index['archive_bytes'] <= MAX_ARCHIVE_BYTES:
        raise ValueError('TRANSPORT_WHOLE_IDENTITY_OR_LIMIT')
    if type(index['archive_file']) is not str or Path(index['archive_file']).name != index['archive_file'] or not index['archive_file'].endswith('.zip'):
        raise ValueError('TRANSPORT_ARCHIVE_NAME')
    parts = index['parts']
    if type(parts) is not list or not 1 <= len(parts) <= 16:
        raise ValueError('TRANSPORT_PART_COUNT_LIMIT')
    names, total = set(), 0
    for part in parts:
        if type(part) is not dict or set(part) != {'file', 'sha256', 'bytes'}:
            raise ValueError('TRANSPORT_PART_SCHEMA')
        name = part['file']
        if type(name) is not str or not re.fullmatch(r'part-[0-9]{3}\.bin', name) or name in names:
            raise ValueError('TRANSPORT_PART_NAME')
        if type(part['bytes']) is not int or not 0 < part['bytes'] <= MAX_PART_BYTES or type(part['sha256']) is not str or not HEX.fullmatch(part['sha256']):
            raise ValueError('TRANSPORT_PART_LIMIT_OR_HASH')
        names.add(name); total += part['bytes']
        inspect(index_path.parent / name, part['bytes'], part['sha256'])
    if total != index['archive_bytes']:
        raise ValueError('TRANSPORT_TOTAL_BYTES')
    pending = safe(output.with_name(output.name + '.PENDING'))
    descriptor = os.open(pending, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(descriptor, 'wb') as destination:
        for part in parts:
            inspect(index_path.parent / part['file'], part['bytes'], part['sha256'], destination)
        destination.flush(); os.fsync(destination.fileno())
    # The parts and final whole are each checked. Failed partial files are preserved.
    digest = hashlib.sha256()
    with pending.open('rb') as source:
        for block in iter(lambda: source.read(65536), b''):
            digest.update(block)
    if pending.stat().st_size != index['archive_bytes'] or digest.hexdigest() != expected_sha256:
        raise ValueError('TRANSPORT_REASSEMBLED_IDENTITY_MISMATCH')
    os.link(pending, output)
    pending.unlink()
    return {'schema': 'agi-alpha/raw-evidence-reassembly006/v1', 'archive_sha256': digest.hexdigest(),
            'archive_bytes': index['archive_bytes'], 'parts': len(parts), 'extracted': False, 'executed': False}


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--index', required=True, type=Path)
    parser.add_argument('--output', required=True, type=Path)
    parser.add_argument('--expected-sha256', required=True)
    args = parser.parse_args()
    print(json.dumps(reassemble(args.index, args.output, args.expected_sha256)))
