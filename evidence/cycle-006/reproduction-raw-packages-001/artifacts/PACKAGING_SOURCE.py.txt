import hashlib,json,stat,sys,zipfile
from pathlib import Path
root=Path(sys.argv[1]);output=Path(sys.argv[2]);output.mkdir(parents=True,exist_ok=False)
sha=lambda b:hashlib.sha256(b).hexdigest()
reports=[]
for name in ('connected-reproduction-001','connected-reproduction-002'):
 source=root/'evidence/cycle-006'/name
 files=sorted(p for p in source.rglob('*') if p.is_file())
 if any(p.is_symlink() for p in source.rglob('*')):raise ValueError('RAW_SOURCE_SYMLINK')
 rows=[];archive=output/(name+'-complete-raw.zip')
 with zipfile.ZipFile(archive,'x',compression=zipfile.ZIP_DEFLATED,compresslevel=9) as sink:
  for p in files:
   raw=p.read_bytes();relative=str(p.relative_to(source));mode=stat.S_IMODE(p.stat().st_mode)
   info=zipfile.ZipInfo(relative,date_time=(2026,10,29,0,0,0));info.compress_type=zipfile.ZIP_DEFLATED;info.external_attr=(stat.S_IFREG|mode)<<16
   sink.writestr(info,raw,compresslevel=9)
   rows.append({'path':relative,'sha256':sha(raw),'bytes':len(raw),'mode':oct(mode)})
 with zipfile.ZipFile(archive) as zipped:
  if zipped.namelist()!=[r['path'] for r in rows]:raise ValueError('ARCHIVE_FILE_SET_CHANGED')
  for r in rows:
   raw=zipped.read(r['path'])
   if sha(raw)!=r['sha256'] or len(raw)!=r['bytes'] or sha((source/r['path']).read_bytes())!=r['sha256']:raise ValueError('RAW_REPRODUCTION_IDENTITY_MISMATCH')
 manifest={'schema':'agi-alpha/lossless-execution-evidence006/v1','source_directory':str(source.relative_to(root)),'archive_file':archive.name,'archive_sha256':sha(archive.read_bytes()),'archive_bytes':archive.stat().st_size,'files':rows,'all_archived_and_live_source_bytes_rechecked':True,'original_files_modified':False,'interpretation':'Lossless publication packaging of all raw files, including failures and append-only post-run qualification notes. Original execution receipts retain their original timing, source identities and accounting scope.'}
 with (output/(name+'-MANIFEST.json')).open('x') as stream:json.dump(manifest,stream,indent=2);stream.write('\n')
 reports.append({k:manifest[k] for k in ('source_directory','archive_file','archive_sha256','archive_bytes')})
print(json.dumps(reports))
