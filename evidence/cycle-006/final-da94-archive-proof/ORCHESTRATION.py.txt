import hashlib
import io
import json
import os
from pathlib import Path, PurePosixPath
import subprocess
import sys
import tarfile
import tempfile

REPO = Path('/workspace/scratch/681b6e3921b1/lab-cycle006')
OUTPUT = REPO / 'evidence/cycle-006/final-clean-archive-qualification-run/artifacts'
EXPECTED_FREEZE = '43f75e28e5fbdc00f1bc40c506dfb319b1e1a06365f552de9088148acdb0f380'
EXPECTED_COMMIT_PREFIX = 'da94c2e'
EXPECTED_CHECKER = '646dfd9df78a44ca12e1fcc75b751cfb4892583cdaa278db20f8941a7bfdbb3e'

def sha(raw): return hashlib.sha256(raw).hexdigest()

def call(command, cwd, log):
    with log.open('xb') as stream:
        result = subprocess.run(command, cwd=cwd, stdout=stream, stderr=subprocess.STDOUT, timeout=690)
    if result.returncode:
        raise RuntimeError('QUALIFICATION_COMMAND_FAILED '+log.name+' '+str(result.returncode))

OUTPUT.mkdir(parents=True, exist_ok=False)
commit = subprocess.check_output(['/usr/bin/git','rev-parse','HEAD'],cwd=REPO,text=True).strip()
if not commit.startswith(EXPECTED_COMMIT_PREFIX):
    raise ValueError('FINAL_SOURCE_COMMIT_CHANGED '+commit)
export = subprocess.check_output(['/usr/bin/git','archive','--format=tar',commit],cwd=REPO)
with tempfile.TemporaryDirectory(prefix='alpha006-final-source-',dir='/tmp') as temporary:
    source=Path(temporary)/'source'
    if any((p/'.git').exists() for p in [source,*source.parents]):
        raise ValueError('FINAL_SOURCE_GIT_ANCESTRY_PRESENT')
    source.mkdir()
    payload_count=0
    payload_bytes=0
    with tarfile.open(fileobj=io.BytesIO(export),mode='r:') as archive:
        for member in archive:
            name=PurePosixPath(member.name)
            if name.is_absolute() or '..' in name.parts or '.git' in name.parts or '\\' in member.name or name.as_posix()!=member.name.rstrip('/'):
                raise ValueError('GIT_EXPORT_PATH_SCOPE')
            if member.isdir():
                continue
            if not member.isfile():
                raise ValueError('GIT_EXPORT_NONREGULAR_ENTRY')
            raw=archive.extractfile(member).read()
            target=source/member.name
            target.parent.mkdir(parents=True,exist_ok=True)
            with target.open('xb') as stream:stream.write(raw)
            payload_count+=1;payload_bytes+=len(raw)
    del export
    freeze=source/'campaign/cycle-006/REPRODUCTION_FREEZE.json'
    if sha(freeze.read_bytes())!=EXPECTED_FREEZE:
        raise ValueError('FINAL_REPRODUCTION_FREEZE_IDENTITY_MISMATCH')
    if sha((source/'verification/cycle006_archive_check.py').read_bytes())!=EXPECTED_CHECKER:
        raise ValueError('FINAL_ARCHIVE_CHECKER_IDENTITY_MISMATCH')
    provenance={'schema':'agi-alpha/cycle006-final-clean-git-export/v1','source_commit':commit,'git_tree':subprocess.check_output(['/usr/bin/git','rev-parse',commit+'^{tree}'],cwd=REPO,text=True).strip(),'tracked_payload_count':payload_count,'tracked_payload_bytes':payload_bytes,'reproduction_freeze_sha256':EXPECTED_FREEZE,'checker_sha256':EXPECTED_CHECKER,'no_git_ancestry':True,'scope':'Clean exact tracked source export. Untracked raw trees and failed paper temporary directories remain in the original worktree and are not silently removed.'}
    (OUTPUT/'SOURCE_EXPORT.json').write_text(json.dumps(provenance,indent=2)+'\n')
    artifact=OUTPUT/'AGI_ALPHA_Perturbation_Construction_006_source_evidence.zip'
    call([sys.executable,'-I',str(source/'flywheel/archive_v2.py'),'build',str(source),str(artifact)],source,OUTPUT/'build.log')
    expected_sha=sha(artifact.read_bytes())
    (OUTPUT/'SOURCE_ZIP_IDENTITY.json').write_text(json.dumps({'archive_sha256':expected_sha,'archive_bytes':artifact.stat().st_size,'source_commit':commit,'reproduction_freeze_sha256':EXPECTED_FREEZE,'scope':'Identity sealed before the new archive checker imports any extracted source.'},indent=2)+'\n')
    call([sys.executable,str(source/'verification/cycle006_archive_check.py'),'--archive',str(artifact),'--expected-sha256',expected_sha,'--output',str(OUTPUT/'qualification')],source,OUTPUT/'checker.log')
    report=json.loads((OUTPUT/'qualification/REPORT.json').read_text())
    if report['runtime_qualified'] is not True:
        raise RuntimeError('FINAL_CLEAN_ARCHIVE_NOT_QUALIFIED')
    print(json.dumps({'runtime_qualified':True,'source_commit':commit,'archive_sha256':expected_sha,'archive_bytes':artifact.stat().st_size,'report':str(OUTPUT/'qualification/REPORT.json')}))
