import argparse
import hashlib
import io
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tarfile
import time


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--private-output', required=True)
    parser.add_argument('--evidence-output', required=True)
    parser.add_argument('--stop-after-events', type=int, choices=[3, 4], default=4)
    args = parser.parse_args()
    repo = Path(__file__).resolve().parents[1]
    private = Path(args.private_output).absolute()
    evidence = Path(args.evidence_output).absolute()
    if private.exists() or evidence.exists() or repo in private.parents:
        raise ValueError('NEW_PRIVATE_OUTPUT_OUTSIDE_REPOSITORY_AND_NEW_EVIDENCE_OUTPUT_REQUIRED')
    if any(p.is_symlink() for p in [private, *private.parents, evidence, *evidence.parents]):
        raise ValueError('NORMALIZED_OUTPUT_REQUIRED')
    private.mkdir(parents=True, mode=0o700)
    evidence.mkdir(parents=True)
    start, cpu = time.monotonic(), time.process_time()
    legacy = '9facd934d9ae763691e8a51052a67793cac5b48e'
    candidate = 'd2e03f11d145eed908e8d473fc007f8daef6f1df'
    folders = ['adaptive_enterprise', 'connected_enterprise', 'commercial', 'enterprise',
               'experiments', 'vendor', 'flywheel', 'portfolio25', 'portfolio_scale', 'verification']
    for ref, name in [(legacy, 'old'), (candidate, 'new')]:
        root = private / name
        root.mkdir(mode=0o700)
        raw = subprocess.check_output(['git', 'archive', ref, '--', *folders], cwd=repo)
        with tarfile.open(fileobj=io.BytesIO(raw)) as archive:
            for member in archive.getmembers():
                if member.isfile() and member.name.endswith('.py'):
                    relative = Path(member.name)
                    if relative.is_absolute() or '..' in relative.parts:
                        raise ValueError('ARCHIVE_PATH_REQUIRED')
                    target = root / relative
                    target.parent.mkdir(parents=True, exist_ok=True)
                    target.write_bytes(archive.extractfile(member).read())
    old, new = private / 'old', private / 'new'
    sys.path.insert(0, str(new))
    from adaptive_enterprise.instruments import prepare_services
    from commercial.ledger import digest
    from connected_enterprise.finance import JournalAccount, source_identity
    from connected_enterprise.upgrades import authorize_upgrade
    manifest = prepare_services(old, 'services')
    (old / 'MANIFEST.json').write_text(json.dumps(manifest))
    shutil.copytree(old / 'services', new / 'services')
    business = private / 'business'
    code = '''import json,sys
from pathlib import Path
from connected_enterprise.engine import default_config,run_world
root,out=map(Path,sys.argv[1:3]);m=json.load(open(root/'MANIFEST.json'));cfg=default_config();cfg['horizon_days']=12;cfg['customer_pool']=12
r=run_world(root,m,cfg,14001,'growth',out,stop_after_events=int(sys.argv[3]))
print(json.dumps({'record':r['owned_account'],'state':json.load(open(out/'STATE.json')),'config':cfg}))
'''
    done = subprocess.run([sys.executable, '-c', code, str(old), str(business), str(args.stop_after_events)],
                          cwd=old, env={**os.environ, 'PYTHONPATH': str(old)},
                          capture_output=True, text=True, timeout=120)
    (evidence / 'LEGACY_EXECUTION.stderr.log').write_text(done.stderr)
    if done.returncode:
        raise RuntimeError('LEGACY_EXECUTION_FAILED:' + done.stderr[-4000:])
    observed = json.loads(done.stdout)
    record, checkpoint, config = observed['record'], observed['state'], observed['config']
    state = checkpoint['body']['state']
    if not state['queue'] or not record['ledger']['orders']:
        raise AssertionError('REAL_OPERATING_QUEUE_AND_COMMITTED_ORDER_REQUIRED')
    path = business / 'account.sqlite'
    contract = record['ledger']['config']['contract']
    ticket = authorize_upgrade(historical_root=old, target_root=new, manifest=manifest,
                               path=path, opening=config['opening'], working=config['working'],
                               contract=contract, expected_state_digest=record['snapshot']['state_digest'],
                               ident='readonly-review-source-upgrade', day=state['last_time'])
    with JournalAccount.upgrade_source(new, manifest, path, opening=config['opening'],
                                       working=config['working'], contract=contract, authorization=ticket) as account:
        checked = account.audit()
        after = account.export()
        if (not checked['accepted'] or after['ledger']['actions'][:-1] != record['ledger']['actions']
                or after['ledger']['summary'] != record['ledger']['summary']):
            raise AssertionError('AUTHENTICATED_ACCOUNT_UPGRADE_CONTINUITY_REQUIRED')
        try:
            account.verify_checkpoint(checkpoint)
        except ValueError as error:
            rejection = str(error)
        else:
            raise AssertionError('EXPECTED_MISSING_STATE_MIGRATION_GUARD_NOT_REPRODUCED')
        if rejection != 'OPERATING_CHECKPOINT_SOURCE_OR_AUTHORITY_CHANGED':
            raise AssertionError('UNEXPECTED_REJECTION:' + rejection)
    artifacts = {'LEGACY_PUBLIC_RECORD.json': record, 'LEGACY_CHECKPOINT.json': checkpoint,
                 'UPGRADED_PUBLIC_RECORD.json': after, 'SERVICES.json': manifest}
    for name, value in artifacts.items():
        (evidence / name).write_text(json.dumps(value, sort_keys=True, indent=2) + '\n')
    result = {'schema': 'agi-alpha/overnight022-migration-review/v1',
              'scope': 'ENGINEERING_GUARD_REPRODUCTION_NOT_MIGRATION_QUALIFICATION',
              'legacy_source': legacy, 'candidate_source': candidate,
              'legacy_financial_source': source_identity(old), 'candidate_financial_source': source_identity(new),
              'source_upgrade_accepted': checked['accepted'], 'queue_before': state['queue'],
              'processed_events': state['processed_events'], 'legacy_orders': len(record['ledger']['orders']),
              'legacy_summary': record['ledger']['summary'], 'prior_native_prefix_preserved': True,
              'transition_count': len(after['ledger']['actions']) - len(record['ledger']['actions']),
              'checkpoint_restore_error': rejection, 'expected_guard_reproduced': True,
              'wall_seconds': time.monotonic() - start, 'wrapper_cpu_seconds': time.process_time() - cpu,
              'children_cpu_seconds': sum(os.times()[2:4]), 'host_and_model_billing': 'UNKNOWN',
              'private_fixture_retained_outside_repository': True,
              'independence': 'Separate review agent; shared model family, host, repository and data. Candidate unchanged; exposed fixture engineering reproduction.',
              'public_files': {p.name: hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(evidence.iterdir()) if p.is_file()},
              'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
              'claim': 'Owned Account source upgrade preserves monetary history; canonical nonempty operating STATE migration remains guarded and unimplemented.'}
    (evidence / 'RESULT.json').write_text(json.dumps(result, indent=2, sort_keys=True) + '\n')
    print(json.dumps({k: result[k] for k in ['source_upgrade_accepted', 'processed_events', 'legacy_orders', 'checkpoint_restore_error', 'wall_seconds', 'wrapper_cpu_seconds', 'children_cpu_seconds']}, sort_keys=True))


if __name__ == '__main__':
    main()
